## Design

The code handles /0, /31 and /32 addressing as special cases. The code implements strict checks for these addresses to prevent the logic for "normal" addresses from running on them, which would cause errors and crashes. For instance, in cidr_mask, shifting a 32-bit integer by 32 bits (in the case of a /0 address) causes undefined behavior, so instead a check for prefix == 0 is done, and 0 is explicitly returned. 

The table finds the longest prefix by by choosing the route that matches the destination IP and has the largest cidr prefix length. This can be done regardless of the order of the table because take_lookup() scans every possible route and remembers the matching route with the greatest prefix length. As it iterates, the current best matching route will be compared to the current table entry, and if the current entry is a greater match, then it becomes the new best matching route.

```c
bool ip_parse(const char *text, uint32_t *addr)
{
    uint32_t result = 0;
    const char *p;

    if (text == NULL || addr == NULL) {
        return false;
    }

    p = text;

    for (int octet_index = 0; octet_index < 4; octet_index++) {
        unsigned int octet = 0;
        int digits = 0;

        /* Each octet must begin with a decimal digit. */
        if (*p < '0' || *p > '9') {
            return false;
        }

        /* Reject leading zeros, except for the octet "0". */
        if (*p == '0' && p[1] >= '0' && p[1] <= '9') {
            return false;
        }

        /* Parse the octet and reject values greater than 255. */
        while (*p >= '0' && *p <= '9') {
            octet = octet * 10 + (unsigned int)(*p - '0');
            digits++;

            if (octet > 255) {
                return false;
            }

            p++;
        }

        result = (result << 8) | octet;

        if (octet_index < 3) {
            /* The first three octets must end with a period. */
            if (*p != '.') {
                return false;
            }

            p++;
        } else {
            /* No characters may follow the fourth octet. */
            if (*p != '\0') {
                return false;
            }
        }
    }

    *addr = result;
    return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
    if (out == NULL) {
        return;
    }

    snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
             (unsigned int)((addr >> 24) & 0xFF),
             (unsigned int)((addr >> 16) & 0xFF),
             (unsigned int)((addr >> 8) & 0xFF),
             (unsigned int)(addr & 0xFF));
}

bool cidr_parse(const char *text, cidr_t *block)
{
    uint32_t addr;
    uint32_t prefix = 0;
    const char *slash;
    const char *p;
    cidr_t result;

    if (text == NULL || block == NULL) {
        return false;
    }

    /* Find the slash separating the address and prefix. */
    slash = strchr(text, '/');

    if (slash == NULL) {
        return false;
    }

    /* Reject a second slash. */
    if (strchr(slash + 1, '/') != NULL) {
        return false;
    }

    /*
     * Copy the address portion into a temporary buffer.
     * A dotted quad can contain at most 15 characters.
     */
    size_t addr_len = (size_t)(slash - text);

    if (addr_len == 0 || addr_len >= IP_STR_LEN) {
        return false;
    }

    char addr_text[IP_STR_LEN];
    memcpy(addr_text, text, addr_len);
    addr_text[addr_len] = '\0';

    if (!ip_parse(addr_text, &addr)) {
        return false;
    }

    /* The prefix is required and must start with a digit. */
    p = slash + 1;

    if (*p < '0' || *p > '9') {
        return false;
    }

    /* Reject leading zeros, such as /024. */
    if (*p == '0' && p[1] != '\0') {
        return false;
    }

    /* Parse the prefix, which must be between 0 and 32. */
    while (*p >= '0' && *p <= '9') {
        prefix = prefix * 10 + (uint32_t)(*p - '0');

        if (prefix > 32) {
            return false;
        }

        p++;
    }

    /* Reject trailing characters or whitespace. */
    if (*p != '\0') {
        return false;
    }

    result.addr = addr;
    result.prefix = (uint8_t)prefix;

    /* Modify the output only after successful parsing. */
    *block = result;
    return true;
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
    uint32_t added_bits = 0;
    uint32_t temp;
    uint32_t new_prefix;
    uint64_t subnet_size;
    cidr_t result;

    if (out == NULL) {
        return false;
    }

    if (!cidr_is_network(block)) {
        return false;
    }

    if (n == 0 || (n & (n - 1)) != 0) {
        return false;
    }

    if (i >= n) {
        return false;
    }

    /* Count the bits needed to represent n equal-sized subnets. */
    temp = n;

    while (temp > 1) {
        temp >>= 1;
        added_bits++;
    }

    new_prefix = (uint32_t)block.prefix + added_bits;

    if (new_prefix > 32) {
        return false;
    }

    result.prefix = (uint8_t)new_prefix;

    /* Calculate each subnet's size without 32-bit overflow. */
    subnet_size = UINT64_C(1) << (32 - new_prefix);

    result.addr = block.addr + (uint32_t)((uint64_t)i * subnet_size);

    *out = result;
    return true;
}

fwd_table_t *table_create(void)
{
    fwd_table_t *table = malloc(sizeof(*table));

    // GCOVR_EXCL_START
    if (table == NULL) {
        return NULL;
    }
    // GCOVR_EXCL_STOP

    table->routes = NULL;
    table->count = 0;
    table->capacity = 0;

    return table;
}

void table_destroy(fwd_table_t *table)
{
    if (table == NULL) {
        return;
    }

    free(table->routes);
    free(table);
}

size_t table_count(const fwd_table_t *table)
{
    if (table == NULL) {
        return 0;
    }

    return table->count;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
    if (table == NULL || route == NULL) {
        return false;
    }

    /* Validate the destination prefix and network address. */
    if (route->dest.prefix > 32 || !cidr_is_network(route->dest)) {
        return false;
    }

    /* Calculate the new capacity if the table is full. */
    size_t new_capacity = 0;

    if (table->count == table->capacity) {
        if (table->capacity == 0) {
            new_capacity = 4;
        } else {
            if (table->capacity > SIZE_MAX / 2) {
                return false;
            }

            new_capacity = table->capacity * 2;

            if (new_capacity > SIZE_MAX / sizeof(*table->routes)) {
                return false;
            }
        }
    }

    /* Reject duplicate destination networks. */
    for (size_t i = 0; i < table->count; i++) {
        if (table->routes[i].dest.addr == route->dest.addr &&
            table->routes[i].dest.prefix == route->dest.prefix) {
            return false;
        }
    }

    /* Grow the route array if necessary. */
    if (table->count == table->capacity) {
        route_t *new_routes = realloc(
            table->routes,
            new_capacity * sizeof(*table->routes)
        );

        //GCOVR_EXCL_START
        if (new_routes == NULL) {
            return false;
        }
        //GCOVR_EXCL_STOP

        table->routes = new_routes;
        table->capacity = new_capacity;
    }

    /* Add the route. */
    table->routes[table->count] = *route;
    table->count++;

    return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
    char *line = NULL;
    size_t line_capacity = 0;
    size_t line_number = 0;
    ssize_t length;
    bool success = true;

    if (bad_line != NULL) {
        *bad_line = 0;
    }

    if (table == NULL || in == NULL) {
        return false;
    }

    while ((length = getline(&line, &line_capacity, in)) != -1) {
        char *comment;
        char *p;
        route_t route;

        (void)length;
        line_number++;

        /* Remove comments. */
        comment = strchr(line, '#');
        if (comment != NULL) {
            *comment = '\0';
        }

        /* Skip whitespace-only lines. */
        p = line;
        while (isspace((unsigned char)*p)) {
            p++;
        }

        if (*p == '\0') {
            continue;
        }

        /* Parse and add the route. */
        if (!route_parse(p, &route) ||
            !table_add(table, &route)) {
            if (bad_line != NULL) {
                *bad_line = line_number;
            }

            success = false;
            break;
        }
    }

    /* A read error is also a failure. */
    // GCOVR_EXCL_START
    if (ferror(in)) {
        success = false;

        if (bad_line != NULL) {
            *bad_line = 0;
        }
    }
    // GCOVR_EXCL_STOP

    free(line);
    return success;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
    const route_t *best = NULL;
    uint8_t best_prefix = 0;
    size_t i;

    if (table == NULL) {
        return NULL;
    }

    for (i = 0; i < table->count; i++) {
        const route_t *route = &table->routes[i];

        if (!cidr_contains(route->dest, dest)) {
            continue;
        }

        if (best == NULL || route->dest.prefix > best_prefix) {
            best = route;
            best_prefix = route->dest.prefix;
        }
    }

    return best;
}
```

```c

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H
```

```c
#include "lab.h"
#include <inttypes.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>

#ifdef TEST
#define main main_exclude
#endif



int main(int argc, char **argv)
{
    if (argc == 1) {
        printf("Usage: myapp info <address>/<prefix>\n"
               "       myapp split <network>/<prefix> <count>\n"
               "       myapp lookup <table-file> <address>...\n");
        return 0;
    }

    /* Task 4.1: info */
    if (strcmp(argv[1], "info") == 0) {
        if (argc != 3) {
            fprintf(stderr,
                    "myapp: info requires exactly one CIDR argument\n");
            return 1;
        }

        cidr_t block;
        if (!cidr_parse(argv[2], &block)) {
            fprintf(stderr, "myapp: invalid CIDR address: %s\n", argv[2]);
            return 1;
        }

        char address[IP_STR_LEN];
        char mask[IP_STR_LEN];
        char network[IP_STR_LEN];
        char broadcast[IP_STR_LEN];
        char first[IP_STR_LEN];
        char last[IP_STR_LEN];

        ip_format(block.addr, address);
        ip_format(cidr_mask(block.prefix), mask);
        ip_format(cidr_network(block), network);
        ip_format(cidr_broadcast(block), broadcast);
        ip_format(cidr_first_host(block), first);
        ip_format(cidr_last_host(block), last);

        printf("address    %s\n", address);
        printf("prefix     /%u\n", (unsigned)block.prefix);
        printf("netmask    %s\n", mask);
        printf("network    %s\n", network);
        printf("broadcast  %s\n", broadcast);
        printf("first host %s\n", first);
        printf("last host  %s\n", last);
        printf("hosts      %" PRIu64 "\n", cidr_host_count(block));

        return 0;
    }

    /* Task 4.2: split */
    if (strcmp(argv[1], "split") == 0) {
        if (argc != 4) {
            fprintf(stderr,
                    "myapp: split requires a CIDR argument and a count\n");
            return 1;
        }

        cidr_t block;
        if (!cidr_parse(argv[2], &block)) {
            fprintf(stderr, "myapp: invalid CIDR address: %s\n", argv[2]);
            return 1;
        }

        if (!cidr_is_network(block)) {
            fprintf(stderr, "myapp: split address is not a network: %s\n",
                    argv[2]);
            return 1;
        }

        /* Parse count as a positive decimal integer. */
        const char *count_text = argv[3];
        if (*count_text == '\0') {
            fprintf(stderr, "myapp: invalid subnet count: %s\n", argv[3]);
            return 1;
        }

        for (const char *p = count_text; *p != '\0'; ++p) {
            if (*p < '0' || *p > '9') {
                fprintf(stderr, "myapp: invalid subnet count: %s\n",
                        argv[3]);
                return 1;
            }
        }

        errno = 0;
        char *end = NULL;
        unsigned long parsed_count = strtoul(count_text, &end, 10);

        if (errno == ERANGE || end == count_text || *end != '\0' ||
            parsed_count == 0 || parsed_count > UINT32_MAX) {
            fprintf(stderr, "myapp: invalid subnet count: %s\n", argv[3]);
            return 1;
        }

        uint32_t count = (uint32_t)parsed_count;

        /* A valid count must be a power of two. */
        if ((count & (count - 1)) != 0) {
            fprintf(stderr,
                    "myapp: subnet count must be a power of two\n");
            return 1;
        }

        /* Check that the resulting prefix does not exceed 32. */
        uint32_t extra_prefix = 0;
        for (uint32_t n = count; n > 1; n >>= 1) {
            ++extra_prefix;
        }

        if ((uint32_t)block.prefix + extra_prefix > 32) {
            fprintf(stderr,
                    "myapp: split would require a prefix longer than 32\n");
            return 1;
        }

        for (uint32_t i = 0; i < count; ++i) {
            cidr_t subnet;

            if (!cidr_subnet(block, count, i, &subnet)) {
                fprintf(stderr,
                        "myapp: unable to calculate subnet %" PRIu32 "\n",
                        i);
                return 1;
            }

            char formatted[CIDR_STR_LEN];
            cidr_format(subnet, formatted);
            printf("%s\n", formatted);
        }

        return 0;
    }

    /* Task 4.3: lookup */
    if (strcmp(argv[1], "lookup") == 0) {
        if (argc < 4) {
            fprintf(stderr,
                    "myapp: lookup requires a table file and at least one address\n");
            return 1;
        }

        const char *filename = argv[2];
        FILE *in = fopen(filename, "r");

        if (in == NULL) {
            fprintf(stderr, "myapp: cannot open %s: %s\n",
                    filename, strerror(errno));
            return 2;
        }

        fwd_table_t *table = table_create();
        if (table == NULL) {
            fprintf(stderr, "myapp: cannot allocate forwarding table\n");
            fclose(in);
            return 2;
        }

        size_t bad_line = 0;

        if (!table_load(table, in, &bad_line)) {
            if (bad_line != 0) {
                fprintf(stderr, "myapp: %s:%zu: not a valid route\n",
                        filename, bad_line);
            } else {
                fprintf(stderr, "myapp: could not read table file %s\n",
                        filename);
            }

            table_destroy(table);
            fclose(in);
            return 2;
        }

        if (fclose(in) != 0) {
            fprintf(stderr, "myapp: could not close table file %s: %s\n",
                    filename, strerror(errno));
            table_destroy(table);
            return 2;
        }

        int malformed = 0;
        int unreachable = 0;

        for (int i = 3; i < argc; ++i) {
            uint32_t addr;

            if (!ip_parse(argv[i], &addr)) {
                fprintf(stderr, "myapp: invalid address: %s\n", argv[i]);
                malformed = 1;
                continue;
            }

            const route_t *route = table_lookup(table, addr);

            if (route == NULL) {
                printf("%s unreachable\n", argv[i]);
                unreachable = 1;
                continue;
            }

            if (route->gateway != 0) {
                char gateway[IP_STR_LEN];
                ip_format(route->gateway, gateway);

                printf("%s via %s dev %s\n",
                       argv[i], gateway, route->dev);
            } else {
                printf("%s dev %s\n", argv[i], route->dev);
            }
        }

        table_destroy(table);

        if (malformed) {
            return 1;
        }

        if (unreachable) {
            return 3;
        }

        return 0;
    }

    fprintf(stderr, "myapp: unknown command: %s\n", argv[1]);
    printf("Usage: myapp info <address>/<prefix>\n"
           "       myapp split <network>/<prefix> <count>\n"
           "       myapp lookup <table-file> <address>...\n");

    return 1;
}
```

```c
/* Helper: initialize output to sentinel values. */
static void init_route_sentinel(route_t *route)
{
    route->dest.addr = UINT32_C(0x12345678);
    route->dest.prefix = 17;
    route->gateway = UINT32_C(0xABCDEF01);
    memset(route->dev, 'X', sizeof(route->dev));
}

/* Helper: verify that failed parsing leaves output unchanged. */
static void assert_route_unchanged(const route_t *route)
{
    TEST_ASSERT_EQUAL_UINT32(UINT32_C(0x12345678),
                             route->dest.addr);
    TEST_ASSERT_EQUAL_UINT8(17, route->dest.prefix);
    TEST_ASSERT_EQUAL_UINT32(UINT32_C(0xABCDEF01),
                             route->gateway);

    for (size_t i = 0; i < DEV_NAME_LEN; i++) {
        TEST_ASSERT_EQUAL_INT('X', route->dev[i]);

{
```
